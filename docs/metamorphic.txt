-- LEVEL 1: Inference of origin --
-- Metamorphic origin: foliation from directional stress, or solid-state
-- recrystallization in non-foliated samples lacking sedimentary/igneous features.

R1_metamorficas:
IF
    Evidence(foliation = 'yes') AND
    Evidence(directional_stress = 'yes')
THEN
    Declare(Origin(type = 'metamorphic'))

R2_metamorficas:
IF
    Evidence(foliation = 'no') AND
    Evidence(solid_state_recrystallization = 'yes') AND
    Evidence(fossils = 'no') AND
    Evidence(stratification = 'no') AND
    Evidence(fractional_crystallization = 'no')
THEN
    Declare(Origin(type = 'metamorphic'))


-- LEVEL 2: Classification --

-- Slate
R3_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(texture = 'slaty_foliation') AND
    Evidence(grain_size = 'very_fine') AND
    Evidence(protolith = 'shale')
THEN
    Declare(Classification(rock = 'slate'))

-- Marble
R4_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(foliation = 'no') AND
    Evidence(hcl_reaction = 'effervescent') AND
    Evidence(protolith = 'limestone')
THEN
    Declare(Classification(rock = 'marble'))

R5_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(texture = 'massive_saccharoidal_recrystallization') AND
    Evidence(hcl_reaction = 'effervescent')
THEN
    Declare(Classification(rock = 'marble'))

-- Gneiss
R6_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(texture = 'foliated_medium_to_coarse') AND
    Evidence(banding = 'alternating_light_quartzofeldspathic_and_dark_mafic')
THEN
    Declare(Classification(rock = 'gneiss'))

R7_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(texture = 'gneissic_banding') AND
    Evidence(metamorphism_grade = 'intense') AND
    Evidence(protolith IN ['granitic', 'sedimentary'])
THEN
    Declare(Classification(rock = 'gneiss'))

-- Quartzite
R8_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(texture = 'massive_non_foliated') AND
    Evidence(mineralogy = 'recrystallized_interlocking_quartz') AND
    Evidence(mohs_hardness >= 6.5)
THEN
    Declare(Classification(rock = 'quartzite'))

R9_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Evidence(protolith = 'quartz_arenite') AND
    Evidence(metamorphism_type IN ['thermal', 'regional'])
THEN
    Declare(Classification(rock = 'quartzite'))


-- LEVEL 3: Engineering recommendation --

R10_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Classification(rock = 'slate')
THEN
    Declare(Recommendation(use = 'Waterproof roofing tiles, waterproofing
        panels and facade cladding (smooth cleavage planes, very low
        permeability)'))

R11_metamorficas:
IF
    Origin(type = 'metamorphic') AND
    Classification(rock = 'marble')
THEN
    Declare(Recommendation(use = 'Interior architectural finishes, decorative
        cladding, floor slabs and monumental elements (workability,
        mirror-like polish, aesthetics)'))
