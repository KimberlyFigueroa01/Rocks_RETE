-- LEVEL 1: Inference of origin --
-- Igneous origin is attributed if there is evidence of magmatic cooling,
-- ruling out exogenous structures (fossils, stratification).

R1_igneas:
IF
    Evidence(fractional_crystallization = 'yes') AND
    Evidence(fossils = 'no') AND
    Evidence(stratification = 'no')
THEN
    Declare(Origin(type = 'igneous'))

R2_igneas:
IF
    Evidence(crystalline_mosaic_texture = 'yes') AND
    Evidence(fossils = 'no') AND
    Evidence(stratification = 'no')
THEN
    Declare(Origin(type = 'igneous'))

R3_igneas:   
IF
    Evidence(texture = 'glassy_amorphous') AND
    Evidence(fossils = 'no') AND
    Evidence(stratification = 'no')
THEN
    Declare(Origin(type = 'igneous'))


-- LEVEL 2: Classification --
-- Each rock has two alternative paths: (a) by texture + physical property,
-- (b) by compositional definition.

-- Granite
R4_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'phaneritic_porphyritic') AND
    Evidence(color = 'light') AND
    Evidence(quartz = 'rich')
THEN
    Declare(Classification(rock = 'granite'))

R5_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(composition = 'felsic')
THEN
    Declare(Classification(rock = 'granite'))

-- Rhyolite
R6_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'aphanitic') AND
    Evidence(color = 'light') AND
    Evidence(quartz = 'present')
THEN
    Declare(Classification(rock = 'rhyolite'))

R7_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(composition = 'volcanic_equivalent_to_granite')
THEN
    Declare(Classification(rock = 'rhyolite'))

-- Basalt
R8_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'aphanitic_volcanic') AND
    Evidence(color = 'dark') AND
    Evidence(density = 'high')
THEN
    Declare(Classification(rock = 'basalt'))

R9_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(composition = 'mafic_ferromagnesian_rich') AND
    Evidence(quartz = 'absent')
THEN
    Declare(Classification(rock = 'basalt'))

-- Diorite
R10_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'phaneritic') AND
    Evidence(quartz = 'absent') AND
    Evidence(composition = 'intermediate_salt_and_pepper')
THEN
    Declare(Classification(rock = 'diorite'))

R11_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'phaneritic') AND
    Evidence(color = 'intermediate') AND
    Evidence(dominant_minerals = 'plagioclase_amphibole')
THEN
    Declare(Classification(rock = 'diorite'))

-- Andesite
R12_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'aphanitic_fine_grained') AND
    Evidence(color = 'medium_gray') AND
    Evidence(quartz = 'not_dominant')
THEN
    Declare(Classification(rock = 'andesite'))

R13_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(composition = 'intermediate_volcanic') AND
    Evidence(phenocrysts = 'plagioclase_amphibole')
THEN
    Declare(Classification(rock = 'andesite'))

-- Gabbro
R14_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'phaneritic') AND
    Evidence(color = 'dark_green_to_black') AND
    Evidence(dominant_minerals = 'ferromagnesian') AND
    Evidence(quartz = 'absent')
THEN
    Declare(Classification(rock = 'gabbro'))

R15_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(composition = 'mafic_plutonic') AND
    Evidence(dominant_minerals = 'pyroxene_calcic_plagioclase')
THEN
    Declare(Classification(rock = 'gabbro'))

-- Obsidian
R16_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'glassy_amorphous') AND
    Evidence(cooling = 'instantaneous')
THEN
    Declare(Classification(rock = 'obsidian'))

R17_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(fracture = 'conchoidal_perfect') AND
    Evidence(luster = 'glassy')
THEN
    Declare(Classification(rock = 'obsidian'))

-- Pegmatite
R18_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(texture = 'pegmatitic') AND
    Evidence(crystal_size_cm > 2)
THEN
    Declare(Classification(rock = 'pegmatite'))

R19_igneas:
IF
    Origin(type = 'igneous') AND
    Evidence(giant_crystals = 'quartz_and_feldspar') AND
    Evidence(environment = 'late_magmatic')
THEN
    Declare(Classification(rock = 'pegmatite'))


-- LEVEL 3: Engineering recommendation --

R20_igneas:
IF
    Origin(type = 'igneous') AND
    Classification(rock = 'basalt')
THEN
    Declare(Recommendation(use = 'Crushed stone aggregate for high-strength
        concrete and asphalt pavements (homogeneous, isotropic, uniform
        stress distribution)'))

R21_igneas:
IF
    Origin(type = 'igneous') AND
    Classification(rock = 'granite')
THEN
    Declare(Recommendation(use = 'Ornamental slabs, paving stones, building
        cladding and heavy foundation blocks (high hardness, low porosity,
        high compressive strength)'))
